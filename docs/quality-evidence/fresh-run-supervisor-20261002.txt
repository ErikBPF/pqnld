def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def listening():
    return any(int(f[1].split(':')[1], 16) == PORT and f[3] == '0A'
               for p in ('/proc/net/tcp', '/proc/net/tcp6')
               for f in (line.split() for line in Path(p).read_text().splitlines()[1:]))

def listener_owned_by(pid):
    sockets = set()
    for fd in Path(f'/proc/{pid}/fd').iterdir():
        try:
            sockets.add(os.readlink(fd))
        except FileNotFoundError:
            pass  # A transient accepted HTTP socket may close during inspection.
    return any(f[1] == f'0100007F:{PORT:04X}' and f[3] == '0A' and f'socket:[{f[9]}]' in sockets
               for f in (line.split() for line in Path('/proc/net/tcp').read_text().splitlines()[1:]))

def get(url):
    budget = min(5, deadline - time.monotonic()) if deadline is not None else 5
    assert budget > 0, 'attempt_deadline'
    try:
        with opener.open(url, timeout=budget) as response:
            assert response.status == 200
            body = response.read()
            if url == ENGINE + '/health':
                health['successful_gets'] += 1
                if health['before_status'] is None:
                    health['before_status'] = response.status
            return body
    except Exception as exc:
        if url == ENGINE + '/health':
            health['failed_gets'] += 1
            health['last_failure_status'] = getattr(exc, 'code', None)
        raise
